Code: [Auswählen]
char* VidMem = (char*)0x0B8000;
void printf( char* t );
void clrscr();


int main() {
char Text[] = "ABC";

clrscr();
printf( Text );

return 0;
}


void printf( char* t ) {
while( *t ){
*VidMem = *t;
t++; VidMem++;
*VidMem = 0x07;
VidMem++;
}
}

VidMem = (char*) 0x0B8000;
int i = 0;

while( i < 4000 ) {
VidMem[i++] = 0;
}
