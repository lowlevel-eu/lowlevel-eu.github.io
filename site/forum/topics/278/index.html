 char* zahl = "123";
  int ergebnis = 0;
  char* p2z = &zahl[0];

  while (*p2z)
  {
    int add = *p2z - '0';  /// Berechne den WErt der Ziffer
    ergebnis*= 10;         // Eine weitere Stelle hintendran
    ++p2z;  
    ergebnis += add;  
  }
