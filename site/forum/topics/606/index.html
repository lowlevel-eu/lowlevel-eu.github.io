Code: [Auswählen]
EnableA20Gate:

.0:
push eax
.1:

in al, 0x64
test al, 00000010b
jnz .1

.2:
mov al, 0xD0
out 0x64, al

.3:
in al, 0x64
test al, 00000001b
jnz .3

.4:
in al, 0x60
or al, 00000010b
push eax

.5:
in al, 0x64
test al, 00000010b
jnz .5

.6:
mov al, 0xD1
out 0x64, al

.7:
in al, 0x64
test al, 00000010b
jnz .7

.8:
pop eax
out 0x60, al

.9:
pop eax
ret
