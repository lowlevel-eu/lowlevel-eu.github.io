sollte net so schwirig sein... bin auch gerade an einem fat16 loader... :
Code: [Auswählen]
;If we aren't booting from a floppy the the mbr-code
;save the mbr-entry to si (pointer) -> starting sector si + 8
.CalculateOffset:
cmp dl, 0x80
jb .GetDiskParams
mov eax, DWORD [si + 8]
mov [PartitionStart], DWORD eax
