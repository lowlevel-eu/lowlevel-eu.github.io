video.c :
#include <video.h>

void ClearScreen(char Color)
{
    VideoPos = 0;

    while(VideoPos < (80 * 25 * 2))
    {
        VideoMem[VideoPos] = ' ';
        VideoPos++;
        VideoMem[VideoPos] = Color;
        VideoPos++;
    };
    VideoPos = 0;
};

unsigned int PrintMsg(char *Msg, char Color)
{
    while(*Msg)
    {
        if(*Msg == '\n')
        {
            VideoPos = (((VideoPos / 160) + 1) * 160) + (VideoPos % 160);
        }
        else
        {
            VideoMem[VideoPos] = *Msg;
            *Msg++;
            VideoPos++;
            VideoMem[VideoPos] = Color;
            VideoPos;
                };
        };

        return(1);
};
