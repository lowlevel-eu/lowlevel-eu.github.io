Blöde Frage zum CommOS
Hi,

genau!

Code: [Auswählen]
gcc -ffreestanding -c -Os -o kernel.obj kernel.c

nasm -f aout -o kernel32.obj kernel32.asm

Und dann Linken:

Code: [Auswählen]
ld -T link.txt -o kernel.bin

Linkerscript:

Code: [Auswählen]
OUTPUT_FORMAT("binary")
INPUT(kernel32.obj kernel.obj)
ENTRY(start)
SECTIONS
{
  .text  0x10000 : {
    code = .; _code = .; __code = .;
    *(.text)
    . = ALIGN(1);
  }
  .data  : {
    data = .; _data = .; __data = .;
    *(.data)
    . = ALIGN(1);
  }
  .bss  :
  {
    bss = .; _bss = .; __bss = .;
    *(.bss)
    . = ALIGN(1);
  }
  end = .; _end = .; __end = .;
}
