Probleme mit Bootloader
1. machn org davor.
2. mach den jump weg, wozu soll der sein?
3. setze prophylaktisch bits auf 16.
4. lade den kernel an 0x50:0x0000, alles andere is platzverschwendung (du musst das org im kernel dann auch auf [org 0x500] statt [org 0x10000] stellen!)
Code: [Auswählen]

[ORG 0x7C00]      ;startup
[BITS 16]         ;16 bit  

start:
mov ax,cs
mov ds,ax
mov es,ax

init_drive:
mov ah,0
mov dl,0
int 13h

jc init_drive

read:
mov ax,50h
mov es,ax
mov bx,0

mov ah,2
mov al,5
mov ch,0
mov cl,2
mov dh,0
mov dl,0
int 13h

jc read

jmp 50h:0000

times 510-($-$$) db 0
dw 0AA55h
