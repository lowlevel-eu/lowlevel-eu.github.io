void kprintf(char *fmt,...)
{
ushort *vid_buf=(ushort*)0xB8000;

fmt=str;

while(*fmt)
{
switch(*fmt)
{
case '\n':
pos+=80-(pos%80);
break;

default:
if(pos>=80*25)
{
for(i=1;i<24;i++)
memcpy(&vid_buf[(i-1)*80],&vid_buf[i*80],160);

}

vid_buf[pos++]=*fmt | 0x0700;

}

}
