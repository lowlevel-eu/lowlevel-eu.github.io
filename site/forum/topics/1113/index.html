Fehler bei Textausgabe durch Bochs...
Hier der Mini Kernel, "msg" ist die oben beschriebene Nachricht.

Code: [Auswählen]
mov ax, 1000h
mov ds, ax
mov es, ax

start:

mov si, msg
call putstr

msg db "Welcome to my OS",13,10,0

putstr:
lodsb
or al, al
jz short putstrd
mov ah, 0x0E
mov bx, 0x007
int 0x10
jmp putstr

putstrd:
retn
