mov si,PartitionTable  ;We want to write fist the partition Table
mov cx,32              ;We want to transfer 32 words

;The next 16 are reserved for the LBA packet
add di,0x0500           ;Load the address again
add di,80               ;We add 80 because PartitionTable = 64 bytes LBA packet = 16 bytes = 80 Bytes
mov byte[di],dl         ;Store the Bootdrive

mov di,0x0600           ;Our C0ode we will write to the Address 0x0600
mov si,ExecutableCode   ;
mov cx,EndOfCode-ExecutableCode
rep movsb               ;Write the Code to the right Address
jmp 0x0000:0x0600       ;Let's go


ExecutableCode:
mov di,0x0500      ;Load the Address of our Structures
mov cl,0           ;In cl we store the number of our Attempts

.NextPartition:
add di,16      ;One entry in the Partition Table is 16 bytes width so we add that value to get the next entry
inc cl         ;We increase our Attempts
cmp cl,3       ;If we have read 4 Partition Table entrys we must quit
ja .NoPartitionBootable ;We only have 4 entry in the Partition Table so we must quit
.VerifyBootability:
mov al,0x80     
cmp al,byte[di]  ;We try if the Partition is bootable so we compare the first byte of the entry with 0x80
jne .NextPartition ;okay it's not bootable so try the next Partition

;Yeah it's bootable first we will try the int 13h extension support
mov ax,0x4100    ;We use BIOS funktion 41h for that
mov bx,0x55AA    ;Setup the other Parameters
mov dl,byte[0500h+80] ;get the Bootdrive from our Structures
int 13h          ;Make the Call
jc .NoPartitionBootable ;If int 13h extension is not supported we quit

;int 13h extension is supported so we setup the lba packet
mov si,di        ;Get the address of the partition entry which is bootable
mov di,0x0500    ;Get the address of our structures
add di,64        ;Set the Address to the begin of our LBA Packet
mov byte[di],16 ;We setup the Size of the Packet its 16 bytes
;mov byte[di+1],0 This byte will be ignored anyway so we don't need to set it
mov word[di+2],1 ;We only read one sector its the bootsektor
mov dword[di+4],0x00007c00 ;We set the low 16 bits to 0x7c00 and the high 16 bits to 0x0000 so we will load it to the correct address

mov ax,word[si+8]  ;in si we stored the Value of the Partition wich we will use we store the low 16 bits of our lba address in ax
mov bx,word[si+10] ;We store the high 16 bits in bx

mov word[di+8],ax  ;First we write the low 16 bits in the packet
mov word[di+10],bx ;Then we store the high 16 bits in the packet

mov dword[di+12],0 ;We don't use the high lba address

push si                     ;Push the address of our partition entry that we will store later in di
mov ax,0x4200
mov dl,byte[0500h+80] ;get the Bootdrive from our Structures
mov si,0500h+64        ;Get the address of our packet
int 13h              ;Load it to the right address
pop di             ;Set es:di to the partition entry which we use
mov dl,byte[0500h+80] ;get the Bootdrive from our Structures
jmp 0x0000:0x7c00    ;Jump to the bootsektor

Re: Von Festplatte booten
Hallo,

ob die Festplatte 4k-Sektoren hat oder nicht ist irrelevant (nach außen exportiert sie aus Kompatiblitätsgründen 512-Byte-Sektoren). Und dass du dir eine 2 TB-Platte aus Jux zum Spielen gekauft hast, nehm ich dir so nicht ab. Da reicht auch der 32 MB USB-Stick, der in der Schublade ganz unten gelegen hat.

Hier ein paar Tips:
  • Nimm Linux, da funktioniert ein "sudo dd if=/bla/blub/mbr.bin of=/dev/sdc".
  • Mach die mbr.bin nur 448 Byte groß, das wird dir später Ärger und versehentlich verschwundene Dateisysteme ersparen.
  • Gucke mit einem Hexeditor auf den ersten Sektor der Festplatte, ob dein MBR wirklich drauf ist. (Nimm dafür ein Linux-Livesystem, wenn du es nicht installieren willst.) Achte auf eventuelle Strings und die Signatur.
  • Achte darauf, dass Partitionen vorhanden sind und exakt eine davon aktiv ist.
  • Nicht jedes BIOS ist dumm. Manche sind intelligenter, als sie sein sollten:
    • Bei einem alten IBM Thinkpad wollte das BIOS nur booten, wenn eine FAT-Partition (Typ 0x06 oder 0x0b) vorhanden war. Es kann gut sein, dass dein BIOS auf die Existenz von Partitionen prüft, ehe es den MBR ausführt; besonders beim Boot von USB sind da vielleicht striktere Prüfungen möglich.
    • Aus der Partitionstabelle ergibt sich ein CHS-Mapping für den int13. Manche BIOSse booten von USB nur mit bestimmten Geometrien ("ZIP-Drive") und ohne Partitionstabelle passt die Geometrie dann nicht.
    [li]Teste den Boot von der Platte mit einem garantiert funktionierendem Bootsektor (Syslinux, DOS, ...)
  • Nimm eine kleinere Festplatte oder einen kleineren USB-Stick zum Testen, bevorzugt kleiner als 504 MB (dann sind alle CHS-Geometrien(*) gleich, aber nicht ZIP-Drive-kompatibel).
  • Lerne deine Tools kennen. Irgendwelche Frontends sind gut, aber du solltest wissen, was dahinter passiert. Betrifft Qemu.

(*) Auch wenn du kein CHS benutzt, kann das eine Rolle spielen.

Was den MBR-Code angeht, so kann man normalerweise einen JMP an die gleiche Stelle machen, um unabhängig von der BIOS-Ladeadresse zu sein - bevor man irgendwelche relativen Adressierungen ausführt. Den ausführbaren Code irgendwo hinzukopieren wirkt jedenfalls unelegant und kostet Platz. Wenn das BIOS den ganzen Sektor lädt, kannst du die Partitionstabelle in-place auslesen, tut es das nicht, musst du ohnehin den ersten Sektor von der Platte lesen.

Re: Von Festplatte booten
Zitat von: Dezimiert am 04. September 2012, 13:32
Allerdings verstehe ich nicht warum den mbr nur 448 bytes groß? Ich dachte das die bytes nach den 448 bytes sowieso vom BIOS ausgefüllt werden, also wo ist der Unterschied dazu ob ich diese Bytes noch benutze oder nicht?
Nein, da wird nichts vom BIOS ausgefüllt. Das BIOS liest den ersten Sektor nach 0x7C00 und überpüft gegeben falls noch die Bootsignatur bevor es die Kontrolle an den Boot Sektor (MBR) übergibt. Mehr tut es (diesbezüglich) nicht. Ab 448 steht (für gewöhnlich) die Partitionstabelle, aber die sollte das BIOS  auch nicht interessieren. Sie wird da nur von z. B. Windows und Partitionierungs tools erwartet. Wenn du sie nicht überschreibst musst du dir nicht die Mühe machen eine Korrekte Tabelle per hand zu erstellen, und zerstörst auch keine potentiell vorhanden Partitionen.

Zitat
Außerdem sollte doh bei 512 Bytes die Bootsignatur auch beim MBR stehen oder?
Ja, aber das steht da im Normalfall schon. Die Text-Ausgabe "Device not Bootable" oder ähnlich stammen meist vom MBR bzw. Bootsector des jeweiligen Gerätes selbst.
Zitat
Und wie finde ich heraus wo eine Partition anfängt, das heißt wo ich dann den eigentlich bootloader hinkopieren soll?
Das kannst du weiter aus der Partitionstabelle ablesen oder vorhanden Tools überlassen.

Aber wenn du auf der Platte so wie so nichts anders drauf hast, kannst du den MBR mit Partitionstabelle auch auslassen. Und die gesamte Platte als eine große Partition oder Diskette betrachten. Das kann dann halt sein das Windows da ein eventuell vorhandenes Dateisystem nicht mehr erkennt. Aber bei einem self-made Bootloader wird es das wahrscheinlich eh nicht geben.

Re: Von Festplatte booten
Hallo,

woher sollte das BIOS denn eine Partitionstabelle generieren, wenn die nicht auf der Festplatte steht? Die steht im ersten Sektor und damit ist gut.

Wenn /dev/sdc deine Festplatte unter Linux ist, dann sind /dev/sdc1, /dev/sdc2, /dev/sdc3 und /dev/sdc4 die primären Partitionen, die dort verzeichnet sein können. Und du solltest dir vielleicht überlegen, mit einem Dateisystem anzufangen. Die meisten Dateisysteme (FAT eingeschlossen) haben reservierte Sektoren, in denen Bootcode liegen kann. Dieser Bootcode kann dann den Kernel, der als gewöhnliche Datei im Dateisystem liegt, dort wieder auslesen.
