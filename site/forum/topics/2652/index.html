hFile = CreateFile(myFile,GENERIC_WRITE,FILE_SHARE_WRITE,0,OPEN_EXISTING,FILE_ATTRIBUTE_NORMAL,NULL);  //Diskettenlaufwerk zum Schreiben öffnen
