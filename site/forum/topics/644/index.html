EDIT: Die Funktion sollte in ax 1 schreiben falls a20 aktiviert ist, sonst 0.

Code: [Auswählen]
;===============================================================
;|||||||||||||||||||||||||||||||||||||||||||||||||||||||||||||||
;|||||||||||||||||||||||||||||||||||||||||||||||||||||||||||||||

;-----------Test if a20 is set----------------------------------
TestA20:
.0:
push bx
push cx
push es

.1:
mov bx, 0xFFFF
mov es, bx
push bx
mov ax, [es:0x100]
not ax
xor bx, bx
mov es, bx
mov [es:0x100], ax
pop bx
mov es,bx
cmp ax, [es:0x100]
je .2

mov ax, 1
jmp .3

.2:
mov ax, 0

.3:
pop es
pop cx
pop bx
ret
